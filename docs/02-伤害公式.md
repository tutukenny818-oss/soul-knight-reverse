# 伤害计算机制 —— 完整还原

> 数据来源：`libil2cpp.so` 反汇编（IFix fallback 分支的原始实现）
> 方法：绕过 IFix 桩，直接读 `tbz w0,#0,<FALLBACK>` 的目标地址

---

## 0. 核心结论（先看这个）

**游戏伤害计算不是一条公式，而是「基础伤害 → 逐级修正 → 取整」的管线。**

```
最终伤害 = max(0, round_half_even( max(倍率,0) × 基础伤害 ))
```

其中「基础伤害」和「倍率」在管线的不同阶段被多组修正函数改写。

## 1. ★★★★★ 最终伤害公式（`CalculateFinalDamage`）

**地址**：`0x0666213C`，fallback @ `0x0666218C`

**完整反汇编**：

```asm
0x0666218C  cbz   x19, #0x666219c      ; this == NULL → 返回 0
0x06662190  cbz   x20, #0x66621a4      ; 倍率对象 == NULL → ratio = 1.0
0x06662194  ldr   s0, [x20, #0x28]     ; ratio = 倍率对象[0x28]
0x06662198  b     #0x66621a8
0x0666219C  mov   w0, wzr              ; return 0
0x066621A0  b     #0x666227c
0x066621A4  fmov  s0, #1.00000000      ; ratio = 1.0
0x066621A8  ldr   s1, [x19, #0x3c]     ; base = this[0x3C]
0x066621B4  fmov  s2, wzr
0x066621B8  fmax  s8, s0, s2           ; m = fmaxf(ratio, 0.0)
0x066621BC  scvtf s9, s1               ; base_f = (float)base
0x066621E0  fmul  s9, s8, s9           ; v = m * base_f
0x066621F4  fcvt  d8, s9               ; d = (double)v
0x06662200  bl    #0x6231a14           ; ← 取整（library round）
; —— 银行家舍入（round half to even）——
0x06662204  fcmp  s9, #0.0
0x06662208  b.ge  #0x6662224           ; v >= 0 → 正向分支
;  负重分支
0x0666220C  fmov  d1, #-0.50000000
0x06662210  fcmp  d0, d1
0x06662214  b.ne  #0x666224c           ; 非 -0.5 → ceil(d-0.5)
0x06662218  ldr   d0, [sp, #0x18]
0x0666221C  fmov  d1, #-1.00000000
;  正分支
0x06662224  fmov  d1, #0.50000000
0x06662228  fcmp  d0, d1
0x0666222C  b.ne  #0x6662258           ; 非 0.5 → floor(d+0.5)
0x06662230  ldr   d0, [sp, #0x18]
0x06662234  fmov  d1, #1.00000000
;  偶数/奇数修正
0x06662238  fcvtzs x8, d0
0x0666223C  fadd   d1, d0, d1
0x06662240  tst    x8, #1              ; 奇偶判定
0x06662244  fcsel  d0, d0, d1, eq      ; 偶数不动，奇数 ±1
;  非 .5 情况
0x0666224C  fadd   d0, d8, d1
0x06662250  frintp d0, d0              ; ceil
0x06662258  fadd   d0, d8, d1
0x0666225C  frintm d0, d0              ; floor
;  钳位到 int 范围 + 非负
0x06662260  mov    x8, #0x7ff0000000000000   ; +INF
0x0666226C  fcmp   d0, d1
0x06662268  fcvtzs w9, d0
0x06662270  mov    w8, #-0xffffffff80000000  ; INT_MIN
0x06662274  csel   w8, w8, w9, eq
0x06662278  bic    w0, w8, w8, asr #31        ; ★ 清掉符号位 = max(0, x)
0x06662288  ret
```

**还原的 C 伪码**：

```c
int CalculateFinalDamage(DamageObject* self, DamageRatio* ratioObj)
{
    if (self == NULL) return 0;

    float ratio = (ratioObj == NULL) ? 1.0f
                                     : *(float*)(ratioObj + 0x28);

    int   baseRaw = *(int*)(self + 0x3C);
    float base    = (float)baseRaw;

    float m = fmaxf(ratio, 0.0f);          // 倍率下限 0
    float v = m * base;

    // —— 银行家舍入（round half to even）——
    double d = (double)v;
    if (v >= 0) {
        if (round(d) == 0.5)  d += (fmod(d,2)!=0) ? 1.0 : 0.0;  // 奇数进位
        else                  d  = floor(d + 0.5);
    } else {
        if (round(d) == -0.5) d -= (fmod(d,2)!=0) ? 1.0 : 0.0;
        else                  d  = ceil(d - 0.5);
    }
    return max(0, (int)d);                  // bic 清符号位
}
```

**关键点**：
| 项 | 值 | 说明 |
|---|---|---|
| 基础伤害字段 | `self + 0x3C` | int32 |
| 倍率字段 | `ratioObj + 0x28` | float32 |
| 倍率缺失时 | `1.0` | 不改变伤害 |
| 舍入方式 | **round half to even** | 银行家舍入（0.5 → 就近偶数） |
| 下限 | `0` | `bic w0,w8,w8,asr#31` |

## 2. ★★★★★ 非暴击减伤（`ApplyNonCriticalDamageReduction`）

**地址**：`0x0647990C`，fallback @ `0x06479954`

```asm
0x06479954  fcmp  s9, #0.0
0x06479958  b.ls  #0x6479984           ; 减伤率 <= 0 → 跳过
0x0647995C  fmov  w8, s9
0x06479960  mvn   w8, w8
0x06479964  tst   w8, #0x7f800000      ; 检查是否 NaN/Inf
0x06479968  b.eq  #0x6479984
0x0647996C  fmov  s0, #1.00000000
0x06479970  fmin  s1, s9, s0           ; s1 = min(减伤率, 1.0)   ← ★ 钳位
0x06479974  fsub  s1, s0, s1           ; s1 = 1.0 - min(减伤率, 1.0)
0x06479978  fcmp  s9, #0.0
0x0647997C  fcsel s0, s1, s0, pl       ; 减伤率 > 0 时用 (1-减伤率)
0x06479980  fmul  s8, s0, s8           ; 伤害 *= 保留率
0x06479990  ret
```

**还原公式**：

```c
float ApplyNonCriticalDamageReduction(float damage, float reduction)
{
    if (reduction <= 0.0f) return damage;          // 无减伤
    if (isnan(reduction) || isinf(reduction)) return damage;

    float keep = 1.0f - fminf(reduction, 1.0f);    // 有效保留率
    return damage * keep;
}
```

**语义**：
| 减伤率 | 结果 |
|---|---|
| ≤ 0 | 原伤害不变 |
| 0 < r < 1 | `伤害 × (1 - r)` |
| ≥ 1 | `伤害 × 0` = 0 |

**注意 `fminf(reduction, 1.0)`**：减伤率被**钳位到 1.0**，所以「减伤 200%」实际等于「减伤 100%」（伤害归零，不会变负）。

## 3. ★★★★ 暴击链

### 3.1 `CalculateCriticalDamage @ 0x06479754`

```
fallback @ 0x6479794:
  ldr x30, [sp, #8]
  fmov s1, wzr                ; 参数 2 = 0.0
  mov v0.16b, v8.16b          ; 参数 1 = 输入伤害
  b  #0x6479838               ; → 主逻辑
```

**内部逻辑**（@0x6479794 起）：

```asm
0x64797E4  cbz  x19, #0x6479828      ; 对象为空 → 返回 1.0
0x64797E8  mov  x0, x19
0x64797EC  mov  w1, wzr
0x64797F0  mov  x2, xzr
0x64797F4  bl   #0x6c8be0c           ; GetValue(obj, 0, null)
0x64797F8  mov  v1.16b, v0.16b
0x64797FC  fcmp s0, #0.0
0x6479800  fmov s0, wzr
0x6479804  b.le #0x647982c           ; 值 <= 0 → 返回 0.0
0x6479808  fmov w8, s1
0x647980C  and  w8, w8, #0x7fffffff  ; 清符号位 = |v|
0x6479810  mov  w9, #0x7f800000      ; +INF
0x6479814  cmp  w8, w9
0x6479818  b.hi #0x647982c           ; NaN/Inf → 返回 0.0
0x647981C  fmov s0, wzr
0x6479820  fcsel s0, s0, s1, eq      ; 有限值 → 用原值
0x6479824  b    #0x647982c
0x6479828  fmov s0, #1.00000000      ; ★ 默认倍率 = 1.0
0x647982C  ret
```

**语义**：

```c
float CalculateCriticalDamage(DamageObject* self, float inDamage)
{
    if (self == NULL) return 1.0f;        // ★ 默认暴击倍率 1.0

    float v = GetValue(self, 0, NULL);
    if (v <= 0.0f) return 0.0f;
    if (isnan(v) || isinf(v)) return 0.0f;
    return v;
}
```

**关键**：
- **暴击倍率默认 1.0**（无暴击时伤害不变）
- 暴击倍率 ≤ 0 → 返回 0.0（暴击反而归零）
- NaN/Inf 保护

### 3.2 暴击修正链的两个入口

| 函数 | 地址 | 说明 |
|---|---|---|
| `ApplyPlayerCriticalToResolvedDamage` | `0x06478B74` | 对「已解析伤害」施暴击 |
| `ApplyPlayerCriticalOnEnemyHit` | `0x06478DA0` | 命中敌人时施暴击 |
| `ApplyCriticalResultDamageModifier` | `0x06477F7C` | 暴击结果伤害修正 |

**共同模式**（两者 fallback 一致）：

```asm
mov x0, x21
mov x1, x19
bl  #0x647861c              ; ★ ResolveWeaponLevel(obj, ...) 先解析武器等级
mov w1, w20                 ; w1 = 暴击标志
mov x2, x19
mov w3, w0                  ; w3 = 武器等级

```

**⟹ 暴击伤害与【武器等级】强相关** —— `ResolveWeaponLevel` 的结果被作为参数传入。

## 4. ★★★★ 伤害管线（综合）

根据已还原的函数，伤害计算的实际顺序：

```
┌─────────────────────────────────────────────────────────┐
│ 1. 基础伤害取值                                          │
│    base = self[0x3C]  (int32)                            │
│    ratio = ratioObj ? ratioObj[0x28] : 1.0               │
├─────────────────────────────────────────────────────────┤
│ 2. 倍率修正（多处，累积）                                │
│    · ApplyAttackDamageMultiplierTalentDelta  (天赋)      │
│    · GetRuntimeDamageBonus                   (运行时)   │
│    · GetAddDamagePercent                     (加成%)   │
│    · GetExtraDamage                          (额外伤害) │
│    · EnhanceBulletDamageFactor               (子弹强化) │
│    · GetScaledConfiguredDamage               (配置缩放) │
├─────────────────────────────────────────────────────────┤
│ 3. 暴击判定（RollCritical / AlwaysCritical）             │
│    命中 → CalculateCriticalDamage → 倍率 v               │
│    · ResolveWeaponLevel 决定暴击倍率                    │
├─────────────────────────────────────────────────────────┤
│ 4. 减伤修正                                              │
│    · ApplyNonCriticalDamageReduction  (非暴击减伤)      │
│    · GetMarkDamageReductionFactor     (标记减伤)        │
├─────────────────────────────────────────────────────────┤
│ 5. 取整输出                                              │
│    CalculateFinalDamage:                                 │
│      最终 = max(0, round_half_even(max(ratio,0)*base))   │
└─────────────────────────────────────────────────────────┘
```

## 8. 方法论突破（重要）

### ★ IFix 桩的 fallback 分支 = 原始实现

**之前误判**：看到 `bl IsPatched` 就认为「逻辑不在静态镜像里」。

**实际情况**：

```asm
bl   IsPatched
tbz  w0, #0, <FALLBACK>    ← ★ 未 patch 时跳这里，就是原始实现！
...
<FALLBACK>:                 ← 原始代码，完整可读
```

**全库统计**：**535 个函数**都有 fallback 分支。

**⟹ 之前认为「被 IFix 覆盖所以读不到」的函数，全部可以还原。**
